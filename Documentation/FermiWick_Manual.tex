\section{Installation and loading}
FermiWick is a paclet. Install it once from its folder (the one with \texttt{PacletInfo.wl}), then load it in every
session, with the Gutzwiller and Fock modules when needed:
\begin{code}
PacletInstall[CreatePacletArchive["/path/to/FermiWick/FermiWick", $TemporaryDirectory]]
Needs["FermiWick`"]; Needs["FermiWick`Gutzwiller`"]; Needs["FermiWick`Fock`"];
\end{code}
To use it without installing, \texttt{PacletDirectoryLoad["/path/to/FermiWick/FermiWick"]} before \texttt{Needs}. The
guided tour, \texttt{Examples.nb} in the repository, runs the examples of the showcase cell by cell.
FermiWick has no dependencies. It is tested with Mathematica 15.0. The test suite runs with
\texttt{math -script Tests/RunTests.wls}.

\section{Conventions}
\begin{itemize}
\item \textbf{Operators.} \fn{c[l1, l2, \dots]} and \fn{cDag[l1, l2, \dots]} annihilate and create a fermion in the mode
  with labels \texttt{l1, l2, \dots}; \fn{DeclareModeLabels} sets the meaning of each label. By default \fn{c[i]} is a
  spinless fermion on site \texttt{i} and \fn{c[i, s]} a fermion on site \texttt{i} with spin \texttt{s}. A site is any
  expression, and two different symbolic sites are different sites: \texttt{KroneckerDelta[i, j]} is $0$. A spin is
  \texttt{SpinUp}, \texttt{SpinDown} or a symbol, split into $\uparrow$ and $\downarrow$ when needed. Generic labels (orbitals,
  momenta, \dots) are compared with \texttt{KroneckerDelta}, which is kept and substituted in what it multiplies.
\item \textbf{Products.} \texttt{a ** b} is the product in the given order; \texttt{a \textperiodcentered{} b} (\texttt{\textbackslash[CenterDot]})
  is the product brought to normal order. Scalars (anything free of operators) factor out.
\item \textbf{Normal order.} All $c^\dagger$ to the left of all $c$; results are expanded sums of monomials.
\item \textbf{Contractions.} On a pair of sites, in spin channels:
  $\langle c^\dagger_{i\sigma}c_{j\sigma'}\rangle=\chi_{ij}\delta_{\sigma\sigma'}+\sum_a\chi^a_{ij}\sigma^a_{\sigma'\sigma}$ and
  $\langle c_{i\sigma}c_{j\sigma'}\rangle=[(\Delta_{ij}+\sum_a\Delta^a_{ij}\sigma^a)\epsilon]_{\sigma\sigma'}$, with $\epsilon_{\uparrow\downarrow}=1$,
  $a=x,y,z$ (written \texttt{"x"}, \texttt{"y"}, \texttt{"z"}): \fn{\textbackslash[Chi][i, j]}, \fn{\textbackslash[Chi][i, j, a]},
  \fn{\textbackslash[CapitalDelta][i, j]}, \fn{\textbackslash[CapitalDelta][i, j, a]}. $\Delta$ is symmetric and $\Delta^a$
  antisymmetric in the sites; Hermiticity gives $\chi_{ji}=\chi^*_{ij}$, $\chi^a_{ji}=\chi^{a*}_{ij}$, and the amplitudes are
  written with the sites in canonical order (\texttt{Conjugate[\textbackslash[Chi][1, 2]]} for $\chi_{21}$). On one site $\chi_{ii}$
  is the density per spin and $\chi^a_{ii}=\langle S^a_i\rangle$. In the general state by spin components,
  $\rho_{i\sigma j\sigma'}=\langle c^\dagger_{i\sigma}c_{j\sigma'}\rangle$ and $\kappa_{i\sigma j\sigma'}=\langle c_{i\sigma}c_{j\sigma'}\rangle$.
\item \textbf{Complex conjugation.} The amplitudes are complex; every other symbol is taken as real by \fn{Dagger}, unless
  declared complex.
\end{itemize}

\section{Gaussian states}
A Gaussian state is an object that gives the contraction $\langle O_1O_2\rangle$ of two ladder operators. It is declared by
its symmetries, or by rules for the contractions. \fn{state["Symmetries"]}, \fn{state["Description"]} and
\fn{state["Contraction"]} give its parts.
\fnentry{GaussianState}{Gaussian states, in which every expectation value follows from the contractions of two operators,
are also called quasi-free states (pure, as Slater or BCS states, or mixed, as Gibbs states of quadratic Hamiltonians).
\fn{GaussianState[symmetries, options]}: the state with the given symmetries, written in spin
channels (Section~2). \fn{GaussianState[]} is the general state. Each symmetry sets channels to zero or makes them real:
\begin{tabular}{ll}\toprule symmetry & what it leaves \\ \midrule
\texttt{"SpinRotation"} & $\chi$, $\Delta$ \\
\texttt{"SpinRotationZ"} & $\chi$, $\chi^z$, $\Delta$, $\Delta^z$ \\
\texttt{"Charge"} & $\chi$, $\chi^a$ (no pairing) \\
\texttt{"TimeReversal"} & $\chi$, $\Delta$ real; $\chi^a$, $\Delta^a$ imaginary; no magnetization \\ \bottomrule\end{tabular}

Options:
\begin{tabular}{lll}\toprule option & default & \\ \midrule
\texttt{"Hopping"}, \texttt{"Pairing"} & $\chi$, $\Delta$ & names of the amplitudes \\
\texttt{"Filling"} & \texttt{Automatic} & $\chi_{ii}$; \texttt{Automatic}: the symbol $\chi_{ii}$ \\
\texttt{"Magnetization"} & \texttt{Automatic} & $\langle S^z_i\rangle$, a vector, or a function of the site \\
\texttt{"OnSitePairing"} & \texttt{Automatic} & $\Delta_{ii}$ \\
\texttt{"Hermitian"} & \texttt{True} & $\chi_{ji}=\chi^*_{ij}$; \texttt{False}: $\chi_{ji}$ is a symbol of its own \\
\texttt{"Rules"} & \texttt{\{\}} & replacements applied to the results \\ \bottomrule\end{tabular}

\fn{GaussianState[\{\{op1, op2\} :> value, \dots\}]} declares the contractions by rules instead, with patterns in the
operators. Pairs with no rule contract to zero, and $\langle c\,c^\dagger\rangle$ follows from $\langle c^\dagger c\rangle$ by the
anticommutator when it has no rule of its own. Two sites with hopping $g$:}

\section{Wick's theorem and mean field}
\fnentry{WickVEV}{\fn{WickVEV[expr, state]} is the expectation value by Wick's theorem: the expression is normal-ordered,
and each monomial is the sum over its pairings with the sign of the permutation. Options: \texttt{"MaxOrder" -> m}, at most
$m$ contractions between different sites; \texttt{"NormalOrder" -> False}, Wick on the expression as it is.}
\example{maxorder}
\fnentry{VerifyNormalOrder}{$0$ when \fn{WickVEV} with and without normal ordering agree.}
\example{verify}
\fnentry{MeanField}{\fn{MeanField[expr, state]} is the Hartree--Fock--Bogoliubov Hamiltonian of \texttt{expr}: every
normal-ordered product $O_1\cdots O_{2n}$ becomes
\[
\sum_{p<q}(-1)^{p+q+1}\,O_pO_q\,\Big\langle\prod_{r\ne p,q}O_r\Big\rangle-(n-1)\,\langle O_1\cdots O_{2n}\rangle ,
\]
two operators kept out in every way and the rest replaced by its expectation value. The result is quadratic, has the same
expectation value as \texttt{expr} and the same first variation with respect to the state: it is the Hamiltonian whose
Gibbs state makes $\langle H\rangle-TS$ stationary among Gaussian states. Quadratic terms are unchanged; odd terms give
zero. The density--density interaction (Hartree and Fock):}
\example{meanfield}

\section{Matrices and exact diagonalisation}
Loaded by \texttt{Needs["FermiWick`Fock`"]}, in a cell of its own (the names must be known before the code that uses them
is read):
\example{fockload}
\textbf{Conventions.} The modes of a space are ladder operators of FermiWick, numbered site by site, then orbital by
orbital, the spin fastest: $\alpha=(\text{site}-1)\,n_o n_s+(\text{orbital}-1)\,n_s+\text{spin}$, the order of
\texttt{Tuples[\{sites, orbitals, spins\}]} and of \fn{ContractionTable}. A basis state is the string of the occupations
$(n_1,\dots,n_M)$, with index $k=\sum_\alpha n_\alpha 2^{M-\alpha}$ (the first mode is the most significant bit): the order of
\texttt{Tuples[\{0, 1\}, M]}. The Jordan--Wigner sign is $c_\alpha|n\rangle=(-1)^{\sum_{\beta<\alpha}n_\beta}|n-e_\alpha\rangle$. The
matrices are \texttt{SparseArray} objects; inside a sector the basis states are in increasing order of $k$.

\fnentry{FockSpace, FockSpaceQ}{\fn{FockSpace[sites]} is the space of spin-$\frac12$ fermions on the sites, with the modes
\fn{c[i, s]}; \fn{FockSpace[sites, "Orbitals" -> \{o1, \dots\}]} has the modes \fn{c[i, o, s]}, and \texttt{"Spins" -> None}
gives spinless modes \fn{c[i]} (or \fn{c[i, o]}); \texttt{"Spins" -> \{s1, s2\}} names the two spins, the first one up.
\texttt{space["Modes"]}, \texttt{space["Count"]} and \texttt{space["Dimension"]} ($2^M$) are its properties.}
\example{fockspace}
\fnentry{ModeIndex}{\fn{ModeIndex[space, c[l1, l2, \dots]]} (or \fn{cDag}) is the index of a mode;
\fn{ModeIndex[space, l1, l2, \dots]} the same from the labels. A mode outside the space gives a message and
\texttt{\$Failed}.}
\example{modeindex}
\fnentry{FockSector, FockSectorQ}{\fn{FockSector[space, "N" -> n, "Sz" -> sz, "Counts" -> \{\{modes, n\}, \dots\}]} keeps
the basis states with $n$ particles, the given $S^z$ (two spins) and $n$ particles in each set of modes (ladder operators
or indices). Every operator is built directly in a sector, without the whole space: sectors with fixed $N$ are enumerated
bit by bit and indexed through the combinatorial number system, so that time and memory scale with their dimension;
sectors without fixed $N$ need the whole space and are allowed up to \texttt{\$FockFullLimit} modes (24).
\fn{ClearFockCache[]} forgets the sectors built so far.}
\example{focksector}
\fnentry{FockMatrix}{\fn{FockMatrix[expr, target]} is the matrix of an expression of FermiWick: ladder operators, local and
composite operators, operators defined with \fn{DefineOperator} (also \fn{GutzwillerProjector}), products with
\texttt{**} in the order written (no normal order is needed), coefficients numbers or symbols. \texttt{target} is a space,
a sector (the image sector is found from the change of $N$, $S^z$ and of every count) or a pair
\texttt{\{fromSector, toSector\}}. The labels of every operator must be those of modes of the space; a symbolic spin
must be summed first (\texttt{Sum[\dots, \{s, \$Spins\}]}).}
\example{fockmatrix}
\example{fockmatrixsymbolic}
\fnentry{FockExpectation}{\fn{FockExpectation[expr, v, sector]} is $\langle v|\,\texttt{expr}\,|v\rangle$ for a normalized
vector \texttt{v} of the sector (or space); an operator that leaves the sector gives $0$. Here the singlet of two sites:}
\example{fockexpectation}
\fnentry{StateToKet}{\fn{StateToKet[target, v]} writes a vector as kets of the local occupations ($0$, $\uparrow$,

\example{basisstate}
\fnentry{SpinSquared}{\fn{SpinSquared[target]} is $\mathbf S^2$ of the whole space or of a sector (two spins), built from

\example{lowest}
\fnentry{LehmannGreen, GreenMatrix}{\fn{LehmannGreen[H, space, modes]} is the single-particle Green's function at zero
temperature, $G_{ab}(z)=\langle c_a(z-H+E_0)^{-1}c^\dagger_b\rangle+\langle c^\dagger_b(z+H-E_0)^{-1}c_a\rangle$, of a Hamiltonian that
conserves $N$ (a matrix on the whole space, or a function sector $\to$ matrix), for the modes given (ladder operators or
indices), averaged over the degenerate ground multiplet: an association with \texttt{"GroundEnergy"},
\texttt{"GroundSectors"}, \texttt{"Degeneracy"}, \texttt{"Poles"} and \texttt{"Residues"}. Options: \texttt{"N"} (the
particle numbers where the ground state is sought, \texttt{All}), \texttt{"UseSz"} (sectors of $S^z$ when $H$ conserves it,
\texttt{Automatic}), \texttt{"Tolerance"} ($10^{-9}$, the width of the ground multiplet), \texttt{"ResidueCutoff"}
($10^{-12}$). The residues add up to the identity (the last entry below). \fn{GreenMatrix[g, z]} evaluates it at a
complex frequency.}
\example{lehmann}
\example{greenmatrix}
\fnentry{ParticleHoleOperator, TimeReversal}{\fn{ParticleHoleOperator[space, first, last]} is the unitary $P$ with
$Pc_\alpha P^{-1}=c^\dagger_\alpha$ for the modes \texttt{first..last} (all by default): $\sigma^x$ on each of them times

for the antiunitary $T=UK$ of spin-$\frac12$ fermions, with \texttt{f[v] = U.Conjugate[v]}.}
\example{particlehole}
\fnentry{SectorBlock}{\fn{SectorBlock[H, sector]} is the block of a matrix on the whole space that acts in the sector.}
